\documentclass[crop,tikz,11pt]{standalone}
\usepackage{geometricfigures}
\usepackage{amsmath,amssymb}
\usetikzlibrary{arrows.meta}

\definecolor{rot}{HTML}{C1701E}
\definecolor{sep}{HTML}{B03030}

% The target decoder with its ripple switched off (c = 0): the latent circle of
% s = |z|^2/2 goes to the circle p_theta = s, on M (radius R = 1 in the q-plane) for s >= s1 and
% off M (R < 1) for s < s1. Drawn in the coordinates (q, p_theta): R is the distance from the axis,
% p_theta the height, oblique projection. Off M = gray (not yellow: no lobe is meant here).
\pgfmathsetmacro{\sone}{1.0}   % s1: below it the decoder leaves M
\pgfmathsetmacro{\ztop}{2.6}   % where the drawn band is cut
\pgfmathsetmacro{\hz}{1.15}    % height scale
\pgfmathsetmacro{\ob}{0.32}    % obliqueness of the circles
% radius of the cap at height s (s < s1), R^2 = s/s1 for the picture
\pgfmathdeclarefunction{Rb}{1}{\pgfmathparse{sqrt(max(0,(#1)/\sone))}}
% projection of the point at angle t (degrees), radius r, height z
\pgfmathdeclarefunction{PX}{2}{\pgfmathparse{(#2)*cos(#1)}}
\pgfmathdeclarefunction{PY}{3}{\pgfmathparse{\hz*(#3)+\ob*(#2)*sin(#1)}}
% a p_theta > 0 rotating orbit of the pendulum
\pgfmathdeclarefunction{pr}{2}{\pgfmathparse{sqrt(2*((#2)-cos(#1)))}}

\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm,>=Stealth,
  back/.style={dash pattern=on 2pt off 2pt,opacity=0.45}]

% ---------------------------------------------------------------- (a) the image of the decoder in (q, p_theta)
\begin{scope}
  % the cap off M: silhouette and fill of the surface of revolution R = Rb(z), 0 <= z <= s1
  \fill[fg!12!bg] plot[domain=0:\sone,samples=40,smooth,variable=\z] ({-Rb(\z)},{\hz*\z})
    -- plot[domain=\sone:0,samples=40,smooth,variable=\z] ({Rb(\z)},{\hz*\z}) -- cycle;
  \fill[fg!12!bg] plot[domain=180:360,samples=60,smooth,variable=\t] ({PX(\t,1)},{PY(\t,1,\sone)})
    -- plot[domain=0:180,samples=60,smooth,variable=\t] ({PX(\t,1)},{PY(\t,1,\sone)}) -- cycle;
  \draw[fg!55!bg] plot[domain=0:\sone,samples=40,smooth,variable=\z] ({-Rb(\z)},{\hz*\z});
  \draw[fg!55!bg] plot[domain=0:\sone,samples=40,smooth,variable=\z] ({Rb(\z)},{\hz*\z});
  % the part on M: the cylinder R = 1 from s1 to the cut
  \fill[fg!3!bg] (-1,{\hz*\sone}) rectangle (1,{\hz*\ztop});
  \draw[fg!55!bg] (-1,{\hz*\sone}) -- (-1,{\hz*\ztop});
  \draw[fg!55!bg] (1,{\hz*\sone}) -- (1,{\hz*\ztop});
  % rims: the circle p_theta = s1, where the decoder joins M, and the cut
  \draw[fg!55!bg,back] plot[domain=0:180,samples=60,smooth,variable=\t] ({PX(\t,1)},{PY(\t,1,\sone)});
  \draw[fg!55!bg] plot[domain=180:360,samples=60,smooth,variable=\t] ({PX(\t,1)},{PY(\t,1,\sone)});
  \draw[fg!55!bg,densely dotted] plot[domain=0:360,samples=90,smooth,variable=\t] ({PX(\t,1)},{PY(\t,1,\ztop)});
  % a rotating orbit with p_theta > 0 on the exact part, H = 2.4 (front solid, back faded)
  \draw[rot,thick,densely dashed] plot[domain=180:360,samples=90,smooth,variable=\t] ({PX(\t,1)},{PY(\t,1,pr(\t,2.4))});
  \draw[rot,thick,densely dashed,back] plot[domain=0:180,samples=90,smooth,variable=\t] ({PX(\t,1)},{PY(\t,1,pr(\t,2.4))});
  % the bottom of the cap: q = 0, the image of the latent origin
  \filldraw[fill=bg,draw=fg,line width=0.45pt] (0,0) circle (1.6pt);
  \node[font=\small,anchor=north] at (0,-0.06) {$q=0$};
  \node[font=\small,anchor=west] at (1.05,{\hz*0.5*(\sone+\ztop)}) {on $\mathcal{M}$};
  \node[font=\small,anchor=west,fg!60!bg] at ({Rb(0.45)+0.08},{\hz*0.4}) {off $\mathcal{M}$};
  \node[font=\small,anchor=east] at (-1.05,{\hz*\sone}) {$p_\theta=s_1$};
  \node[font=\small] at (0,-0.85) {(a)};
\end{scope}

% ---------------------------------------------------------------- (b) its latent chart
\begin{scope}[shift={(4.6,1.45)}]
  % the disk s < s1, decoded off M
  \fill[fg!12!bg] (0,0) circle ({sqrt(2*\sone)*0.62});
  \draw[fg!55!bg] (0,0) circle ({sqrt(2*\sone)*0.62});
  % the image of the band: the annulus s1 <= s <= ztop
  \draw[fg!55!bg,densely dotted] (0,0) circle ({sqrt(2*\ztop)*0.62});
  % the orbit of (a): angle beta = -theta, s = p_theta(theta)
  \draw[rot,thick,densely dashed] plot[domain=0:360,samples=120,smooth,variable=\t]
    ({0.62*sqrt(2*pr(\t,2.4))*cos(-\t)},{0.62*sqrt(2*pr(\t,2.4))*sin(-\t)});
  \filldraw[fill=bg,draw=fg,line width=0.45pt] (0,0) circle (1.6pt);
  \draw[->,fg!50!bg] (-1.75,-1.75) -- (-1.25,-1.75) node[anchor=west,font=\small] {$z_q$};
  \draw[->,fg!50!bg] (-1.75,-1.75) -- (-1.75,-1.25) node[anchor=south,font=\small] {$z_p$};
  \node[font=\small] at (0,-2.3) {(b)};
\end{scope}
\end{tikzpicture}
\end{document}
