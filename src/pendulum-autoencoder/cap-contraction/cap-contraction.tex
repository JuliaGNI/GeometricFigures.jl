% (a) iota~ (the formula of iota for phi in R, 2 pi-periodic) maps the (phi, p_phi) plane onto the cylinder M: an open graph p_phi = -sqrt(2(H - cos phi)) onto a
%     rotating orbit iota o c_R; a closed latent curve onto a loop that bounds its image disk.
% (b) the encoder maps iota o c_R, on the cylinder of (a), to a closed latent curve gamma, which bounds a disk D,
%     and the librating orbit iota o c_L to a small closed curve gamma_L.
% (c) a decoder with Psi^dec o gamma = iota o c_R maps D onto a surface spanning iota o c_R, which cannot lie in M:
%     the cap; gamma_L decodes onto iota o c_L, whose disk can lie in M.

\begin{document}
\begin{tikzpicture}[>=Stealth,
  lab/.style={font=\footnotesize,inner sep=1.5pt}]

% cylinder helper: centre (cx,0), radius 0.9, vertical extent [-1.2,1.2], ellipse ry 0.25
\newcommand{\cylinder}[1]{%
  \fill[fg!5!bg] (#1-0.9,-1.2) -- (#1-0.9,1.2) arc[start angle=180,end angle=360,x radius=0.9,y radius=0.25]
    -- (#1+0.9,-1.2) arc[start angle=0,end angle=-180,x radius=0.9,y radius=0.25] -- cycle;
  \draw[fg!40!bg] (#1-0.9,-1.2) -- (#1-0.9,1.2);
  \draw[fg!40!bg] (#1+0.9,-1.2) -- (#1+0.9,1.2);
  \draw[fg!40!bg] (#1,1.2) ellipse[x radius=0.9,y radius=0.25];
  \draw[fg!40!bg] (#1-0.9,-1.2) arc[start angle=180,end angle=360,x radius=0.9,y radius=0.25];
  \draw[fg!40!bg,dashed] (#1+0.9,-1.2) arc[start angle=0,end angle=180,x radius=0.9,y radius=0.25];}

% ================================================================ (a) iota
\begin{scope}
  \fill[fg!4!bg] (-1.8,-1.3) rectangle (1.8,1.3);
  \node[lab,anchor=north west] at (-1.8,1.3) {$(\tilde\theta,p_{\tilde\theta})\in\mathbb{R}^2$};
  % the open graph p_phi = -sqrt(2(H - cos phi)), H > 1: two periods, highest where cos phi = 1
  \draw[rot,very thick,->] plot[smooth,domain=-1.7:1.7,samples=61,variable=\x] (\x,{-0.6+0.14*cos(\x*360/1.7)});
  \draw[lib,very thick,fill=lib!25!bg] (-0.4,0.55) ellipse[x radius=0.5,y radius=0.3];
\end{scope}
\draw[->,thick,fg!60!bg] (2.0,0) -- node[above,lab,fg] {$\tilde\iota$} (2.9,0);
\begin{scope}[shift={(4.0,0)}]
  \cylinder{0}
  \draw[rot,very thick] (-0.9,-0.4) arc[start angle=180,end angle=360,x radius=0.9,y radius=0.25];
  \draw[rot,very thick,dashed] (0.9,-0.4) arc[start angle=0,end angle=180,x radius=0.9,y radius=0.25];
  \node[lab,rot,anchor=west] at (0.9,-0.45) {$\iota\circ c_R$};
  % c_L near p = 0, above c_R (p < 0), as in the plane of (a)
  \draw[lib,very thick,fill=lib!25!bg] (-0.2,0.45) ellipse[x radius=0.3,y radius=0.17];
  \node[lab,lib,anchor=west] at (0.1,0.4) {$\iota\circ c_L$};
  \node[lab,anchor=south] at (0,1.45) {$\mathcal{M}$};
\end{scope}

% ================================================================ (b) encoder, from the cylinder of (a)
\draw[->,thick,fg!60!bg] (5.1,0) -- node[above,lab,fg] {$\Psi^\mathrm{enc}$} (6.0,0);
\begin{scope}[shift={(7.6,0)}]
  \fill[fg!4!bg] (-1.6,-1.3) rectangle (1.6,1.3);
  \node[lab,anchor=north west] at (-1.6,1.3) {$\mathbb{R}^2$};
  \draw[rot,very thick,fill=rot!15!bg] (0,0) ellipse[x radius=1.1,y radius=0.75];
  \node[lab,rot!70!fg] at (0,0) {$D$};
  \node[lab,rot,anchor=north west] at (0.75,-0.55) {$\gamma$};
  \draw[lib,very thick,fill=lib!25!bg] (1.05,0.95) ellipse[x radius=0.32,y radius=0.2];
  \node[lab,lib,anchor=east] at (0.72,1.0) {$\gamma_L$};
\end{scope}

% ================================================================ (c) decoder: the cap
\draw[->,thick,fg!60!bg] (9.35,0) -- node[above,lab,fg] {$\Psi^\mathrm{dec}$} (10.25,0);
\begin{scope}[shift={(11.4,0)}]
  \cylinder{0}
  % the surface Psi^dec(D): a bowl below c_R, off M (below, so that it leaves c_L above it free)
  \fill[rot!15!bg] (-0.9,-0.4) .. controls (-0.9,-1.25) and (-0.4,-1.65) .. (0,-1.68)

     arc[start angle=0,end angle=180,x radius=0.9,y radius=0.25];
  \draw[rot!60!bg,thin] (-0.9,-0.4) .. controls (-0.9,-1.25) and (-0.4,-1.65) .. (0,-1.68)
     .. controls (0.4,-1.65) and (0.9,-1.25) .. (0.9,-0.4);
  \draw[rot,very thick] (-0.9,-0.4) arc[start angle=180,end angle=360,x radius=0.9,y radius=0.25];
  \draw[rot,very thick,dashed] (0.9,-0.4) arc[start angle=0,end angle=180,x radius=0.9,y radius=0.25];
  \node[lab,rot,anchor=west] at (0.9,-0.45) {$\iota\circ c_R$};
  \node[lab,rot!70!fg] at (0,-1.05) {$\Psi^\mathrm{dec}(D)$};
  \draw[lib,very thick,fill=lib!25!bg] (-0.2,0.45) ellipse[x radius=0.3,y radius=0.17];
  \node[lab,lib,anchor=west] at (0.1,0.4) {$\iota\circ c_L$};
\end{scope}
